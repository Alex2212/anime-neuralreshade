cmake_minimum_required(VERSION 3.25)
project(CelShade VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "CelShade requires Windows x64")
endif()
set(RESHADE_ROOT "${CMAKE_SOURCE_DIR}/external/reshade" CACHE PATH "ReShade 6.1.1 source")
set(TRT_ROOT "${CMAKE_SOURCE_DIR}/external/TensorRT-10.13.3.9" CACHE PATH "TensorRT SDK")
set(CUDA_ROOT "${CMAKE_SOURCE_DIR}/external/cuda-runtime/nvidia/cuda_runtime" CACHE PATH "CUDA runtime SDK")
set(NVRTC_ROOT "${CMAKE_SOURCE_DIR}/external/cuda-nvrtc/nvidia/cuda_nvrtc" CACHE PATH "NVRTC SDK")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_compile_options(/W4 /EHsc /permissive- /utf-8)
add_library(cel_gpu STATIC src/interop/frame_bridge.cpp)
target_include_directories(cel_gpu PUBLIC src)
target_link_libraries(cel_gpu PUBLIC d3d11 dxgi)
# Separate optional inference DLL: missing NVIDIA dependencies cannot prevent the addon loading.
add_library(cel_nvidia SHARED src/inference/nvidia.cpp)
target_include_directories(cel_nvidia PRIVATE src "${TRT_ROOT}/include" "${CUDA_ROOT}/include" "${NVRTC_ROOT}/include")
target_link_directories(cel_nvidia PRIVATE "${TRT_ROOT}/lib" "${CUDA_ROOT}/lib/x64" "${NVRTC_ROOT}/lib/x64")
target_link_libraries(cel_nvidia PRIVATE nvinfer_10 cudart cuda d3d11 dxgi bcrypt)
add_executable(cel_compile_kernels src/tools/compile_kernels.cpp)
target_include_directories(cel_compile_kernels PRIVATE "${NVRTC_ROOT}/include")
add_library(CelShade SHARED src/addon/addon.cpp)
set_target_properties(CelShade PROPERTIES PREFIX "" SUFFIX ".addon64")
target_include_directories(CelShade PRIVATE src "${RESHADE_ROOT}/include" "${RESHADE_ROOT}/deps/imgui")
target_link_libraries(CelShade PRIVATE cel_gpu)
add_executable(cel_benchmark src/tools/benchmark.cpp)
target_link_libraries(cel_benchmark PRIVATE cel_gpu windowscodecs ole32)
add_executable(anime_viewer WIN32 src/viewer/main.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS third_party/nis/NIS_Main.hlsl third_party/nis/NIS_Scaler.h)
file(READ "${CMAKE_SOURCE_DIR}/third_party/nis/NIS_Main.hlsl" NIS_SHADER_SOURCE)
file(READ "${CMAKE_SOURCE_DIR}/third_party/nis/NIS_Scaler.h" NIS_SCALER_SOURCE)
string(REPLACE "#include \"NIS_Scaler.h\"" "${NIS_SCALER_SOURCE}" NIS_SHADER_SOURCE "${NIS_SHADER_SOURCE}")
configure_file(src/interop/nis_shader.hpp.in generated/nis_shader.hpp @ONLY)
target_sources(anime_viewer PRIVATE src/interop/nis_scaler.cpp)
target_include_directories(anime_viewer PRIVATE "${CMAKE_SOURCE_DIR}/third_party/nis" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(anime_viewer PRIVATE cel_gpu d3dcompiler windowsapp dwmapi shell32 windowscodecs ole32)
add_executable(cel_gpu_test tests/gpu_test.cpp)
target_link_libraries(cel_gpu_test PRIVATE cel_gpu)
add_executable(cel_test_host tests/reshade_host.cpp)
target_link_libraries(cel_test_host PRIVATE d3d11 dxgi)
target_include_directories(cel_test_host PRIVATE "${RESHADE_ROOT}/include")
enable_testing()
add_test(NAME gpu_identity COMMAND cel_gpu_test)
add_test(NAME viewer_edge_quality COMMAND anime_viewer --test-quality --report "${CMAKE_BINARY_DIR}/viewer-edge-quality.json")
install(TARGETS CelShade cel_nvidia cel_benchmark anime_viewer RUNTIME DESTINATION .)
install(DIRECTORY shaders/ DESTINATION shaders)
install(DIRECTORY presets/ DESTINATION presets)
install(FILES src/inference/kernels.cu DESTINATION .)
